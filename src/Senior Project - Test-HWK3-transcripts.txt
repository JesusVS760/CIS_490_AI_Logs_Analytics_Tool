        return 0;
    }

    finalPrice = originalPrice;

    
    // Tiered Discount
    
    if (finalPrice >= 300) {
        tieredDiscount = 50;
    }
    else if (finalPrice >= 200 && finalPrice < 300) {
        tieredDiscount = 30;
    }
    else if (finalPrice >= 100 && finalPrice < 200) {
        tieredDiscount = 10;
    }

    if (tieredDiscount > 0) {
        finalPrice -= tieredDiscount;
        cout << "Tiered discount applied: You saved $"

    }
    else {
        cout << "No tiered discount applied." << endl;

    // Red Tag Discount (5%)
   
    cout << "\nDoes the item have a red tag? ";
    cin.ignore();
    getline(cin, redTagResponse);

    if (redTagResponse == "yes" || redTagResponse == "I see one") {
        redTagDiscount = finalPrice * 0.05;
        finalPrice -= redTagDiscount;
        cout << "Red tag discount applied: You saved $" 
             << redTagDiscount << endl;
    }
    else {
        cout << "No red tag discount applied." << endl;

    // Member Discount (15%)
    
    cout << "\nAre you a store member? ";
    getline(cin, memberResponse);

        memberDiscount = finalPrice * 0.15;

        cout << "No member discount applied." << endl;
    }

    
    // Final Output
   
    cout << "\n----- Receipt -----" << endl;
    cout << "Original price: $" << originalPrice << endl;
    cout << "Final price: $" << finalPrice << endl;

    return 0;
}
